  var heroUnit = 'metric';

  function heroSetUnit(u) {
    heroUnit = u;
    var metricBtn   = document.getElementById('hero-unit-metric');
    var imperialBtn = document.getElementById('hero-unit-imperial');
    var metricIn    = document.getElementById('hero-metric-inputs');
    var imperialIn  = document.getElementById('hero-imperial-inputs');
    var active   = 'flex:1;padding:9px 0;border-radius:8px;border:1.5px solid var(--accent);background:var(--accent);color:#fff;font-size:.85rem;font-weight:600;cursor:pointer;font-family:inherit;';
    var inactive = 'flex:1;padding:9px 0;border-radius:8px;border:1.5px solid var(--border);background:transparent;color:var(--ink2);font-size:.85rem;font-weight:600;cursor:pointer;font-family:inherit;';
    if (u === 'metric') {
      metricBtn.style.cssText = active;
      imperialBtn.style.cssText = inactive;
      metricIn.style.display = 'block';
      imperialIn.style.display = 'none';
    } else {
      metricBtn.style.cssText = inactive;
      imperialBtn.style.cssText = active;
      metricIn.style.display = 'none';
      imperialIn.style.display = 'block';
    }
    document.getElementById('hero-result').style.display = 'none';
    document.getElementById('hero-error').style.display = 'none';
  }

  function heroCalc() {
    var bmi, w, h;
    var err = document.getElementById('hero-error');
    var res = document.getElementById('hero-result');
    err.style.display = 'none';
    res.style.display = 'none';

    if (heroUnit === 'metric') {
      w = parseFloat(document.getElementById('hero-weight-kg').value);
      h = parseFloat(document.getElementById('hero-height-cm').value) / 100;
    } else {
      w = parseFloat(document.getElementById('hero-weight-lb').value);
      h = parseFloat(document.getElementById('hero-height-in').value);
      bmi = (w / (h * h)) * 703;
    }

    if (heroUnit === 'metric') bmi = w / (h * h);

    if (!bmi || bmi < 10 || bmi > 80 || isNaN(bmi)) {
      err.style.display = 'block';
      return;
    }

    bmi = Math.round(bmi * 10) / 10;

    var cat, color, hRange;
    if (bmi < 18.5) {
      cat = 'Underweight'; color = '#3A7BBF';
    } else if (bmi < 25) {
      cat = 'Normal weight'; color = '#2D6A4F';
    } else if (bmi < 30) {
      cat = 'Overweight'; color = '#C17B2A';
    } else {
      cat = 'Obese'; color = '#B83232';
    }

    // Healthy weight range for their height
    var hMin, hMax;
    if (heroUnit === 'metric') {
      hMin = Math.round(18.5 * h * h * 10) / 10;
      hMax = Math.round(24.9 * h * h * 10) / 10;
      hRange = 'Healthy range for your height: ' + hMin + ' â€“ ' + hMax + ' kg';
    } else {
      var hIn = parseFloat(document.getElementById('hero-height-in').value);
      hMin = Math.round(18.5 * hIn * hIn / 703 * 10) / 10;
      hMax = Math.round(24.9 * hIn * hIn / 703 * 10) / 10;
      hRange = 'Healthy range for your height: ' + hMin + ' â€“ ' + hMax + ' lb';
    }

    document.getElementById('hero-bmi-val').textContent = bmi;
    var badge = document.getElementById('hero-bmi-badge');
    badge.textContent = cat;
    badge.style.cssText = 'display:inline-block;padding:4px 14px;border-radius:100px;font-size:.82rem;font-weight:600;margin-bottom:10px;background:' + color + '22;color:' + color + ';';
    document.getElementById('hero-healthy-range').textContent = hRange;
    res.style.display = 'block';
  }

  <h2 id="what-does-bmi-measure">What Does BMI Measure?</h2>
  <p>BMI is a simple numerical value derived from a person's weight and height. It does not directly measure body fat, but it serves as a useful screening indicator to identify individuals who may be underweight, at a healthy weight, overweight, or obese.   <p>The concept was developed in the 1830s by Belgian mathematician Adolphe Quetelet and later adopted by the World Health Organization (WHO) as a standard population health metric. Today, it remains one of the most practical tools available for initial weight assessment because it requires only two easily measured values: weight and height.</p>

  <h2 id="what-health-risks-are-associated-with-high-bmi">What Health Risks Are Associated with High BMI?</h2>
  <p>A higher BMI, particularly in the overweight and obese ranges, is associated with an increased risk of several serious health conditions, including:</p>
  <ul>
    <li>Type 2 diabetes</li>
    <li>High blood pressure (hypertension)</li>
    <li>Cardiovascular disease and stroke</li>
    <li>Certain types of cancer</li>
    <li>Sleep apnoea</li>
    <li>Osteoarthritis</li>
    <li>Non-alcoholic fatty liver disease</li>
  </ul>
  <p>It is important to understand that BMI is a risk indicator, not a diagnosis. Many other factors â€” such as diet, physical activity, genetics, and where body fat is stored â€” also influence health outcomes.</p>

  <h2 id="what-are-the-limitations-of-bmi">What Are the Limitations of BMI?</h2>
  <p>While BMI is a useful population-level screening tool, it has well-recognised limitations when applied to individuals:</p>

  <h3>It does not distinguish between muscle and fat</h3>
  <p>BMI cannot tell the difference between lean muscle mass and body fat. A professional athlete or bodybuilder may have a BMI in the overweight range despite having very low body fat. For a more accurate measure of body composition, try our <a href="/body-fat">Body Fat Calculator</a>.</p>

  <h3>It does not account for fat distribution</h3>
  <p>Where fat is stored in the body matters significantly for health risk. Abdominal (visceral) fat â€” stored around the waist and organs â€” carries a higher health risk than fat stored in the hips and thighs. BMI gives no information about fat distribution. Waist circumference measurements can complement BMI for a more complete picture.</p>

  <h3>It varies by ethnicity</h3>
  <p>Research shows that people of Asian descent tend to have higher levels of body fat at lower BMI values compared to people of European descent. Many health organisations now recommend lower BMI thresholds for Asian populations â€” for example, a BMI of 23 or above may be considered overweight for people of East or South Asian heritage. See our guides on <a href="/healthy-bmi-for-women">healthy BMI for women</a> and <a href="/normal-bmi-for-men">normal BMI for men</a> for more context.</p>

  <h3>It does not apply to children in the same way</h3>
  <p>For those under 18, BMI must be interpreted using age- and sex-specific growth charts, as normal body composition changes considerably during childhood and adolescence. Our <a href="/bmi-chart-by-age">BMI Chart by Age</a> covers this in more detail.</p>

  <div class="warning-callout">
    <p><strong>Important:</strong> BMI is a screening tool, not a diagnostic instrument. Always consult a qualified healthcare professional for a complete and personalised health assessment.</p>
  </div>

  <h2 id="is-a-normal-bmi-always-healthy">Is a Normal BMI Always Healthy?</h2>
  <p>Not necessarily. A person can have a BMI within the normal range and still have metabolic risk factors such as high blood pressure, elevated blood sugar, or unfavourable cholesterol levels. This is sometimes referred to as "metabolically obese, normal weight" (MONW).</p>
  <p>On the other hand, some people with a BMI slightly above the normal range may have excellent metabolic health, particularly if they are physically active and have healthy blood markers.</p>
  <p>BMI works best as one data point among many, not as a standalone measure of health. Our <a href="/health-profile">Health Profile</a> tool combines BMI with several other key metrics to give you a more complete picture.</p>

      <thead><tr><th>BMI Range</th><th>Category</th><th>General Health Risk</th></tr></thead>
      <tbody>
        <tr><td>Below 18.5</td><td>Underweight</td><td>Increased risk (malnutrition, bone loss)</td></tr>
        <tr><td>18.5 â€“ 24.9</td><td>Normal weight</td><td>Lowest risk range</td></tr>
        <tr><td>25.0 â€“ 29.9</td><td>Overweight</td><td>Moderately increased risk</td></tr>
        <tr><td>30.0 and above</td><td>Obese</td><td>Substantially increased risk</td></tr>
      </tbody>
    </table>
    <p>BMI's main strength is its simplicity â€” it requires no special equipment, takes seconds to calculate, and is highly predictive of health outcomes across large populations. Its main weakness is that it cannot distinguish between fat mass and muscle mass.</p>
  </div>

  <div class="article-section">

<h2>What Body Fat Percentage Measures</h2>
    <p>Body fat percentage measures the proportion of your total body weight that is fat tissue. Unlike BMI, it directly distinguishes between fat mass and lean mass (muscle, bone, water, and organs).</p>
    <p>Healthy body fat ranges differ significantly by sex and age:</p>
    <table>
      <thead><tr><th>Category</th><th>Women</th><th>Men</th></tr></thead>
      <tbody>
        <tr><td>Essential fat</td><td>10 â€“ 13%</td><td>2 â€“ 5%</td></tr>
        <tr><td>Athletes</td><td>14 â€“ 20%</td><td>6 â€“ 13%</td></tr>
        <tr><td>Fitness</td><td>21 â€“ 24%</td><td>14 â€“ 17%</td></tr>
        <tr><td>Acceptable</td><td>25 â€“ 31%</td><td>18 â€“ 24%</td></tr>
        <tr><td>Obese</td><td>32% and above</td><td>25% and above</td></tr>
      </tbody>
    </table>
    <p>Body fat percentage can be measured through DEXA scans (most accurate), hydrostatic weighing, skinfold calipers, or bioelectrical impedance (common in smart scales, least accurate).</p>
  </div>

  <div class="article-section">
    <h2>Where BMI Falls Short</h2>
    <h3>The "skinny fat" problem</h3>
    <p>A person can have a completely normal BMI â€” say, 22 â€” while carrying a high percentage of body fat and very little muscle. This condition, sometimes called "normal weight obesity" or "skinny fat," carries many of the same metabolic risks as clinical obesity: insulin resistance, elevated triglycerides, and higher cardiovascular risk. BMI alone would miss this entirely.</p>

    <h3>The athlete problem</h3>
    <p>Conversely, a well-trained athlete may have a BMI in the "overweight" or even "obese" range due to high muscle mass, despite having very low body fat and excellent cardiovascular health. BMI cannot distinguish muscle from fat â€” it only sees total weight relative to height.</p>

    <h3>The age problem</h3>
    <p>As people age, they naturally lose muscle mass and gain fat, even if their weight stays exactly the same. An older adult with an "ideal" BMI of 22 may have significantly higher body fat than a younger adult at the same BMI â€” a difference with real health implications that BMI simply cannot capture.</p>
  </div>

  <div class="article-section">
    <h2>Where Body Fat Percentage Falls Short</h2>
    <p>Body fat percentage isn't without its own limitations. Accurate measurement requires either expensive equipment (DEXA scans cost $50â€“$150) or methods that introduce significant error. Consumer smart scales using bioelectrical impedance can be off by 5â€“8 percentage points depending on hydration levels, time of day, and body type.</p>
    <p>Body fat percentage also doesn't tell you <em>where</em> fat is stored â€” and location matters enormously. Visceral fat (fat around the abdominal organs) is far more metabolically dangerous than subcutaneous fat (fat under the skin). Two people with identical body fat percentages can have very different health risks depending on fat distribution.</p>
  </div>

  <div class="article-section">
    <h2>Using Both Together: The Practical Approach</h2>
    <p>The most useful approach for most people is to use BMI as a starting point â€” it's free, instant, and clinically validated â€” and then supplement it with additional measurements if you need a more complete picture.</p>

    <div class="highlight-box">
      <strong>A practical framework:</strong>
      <ol style="margin: 8px 0 0 18px; line-height: 1.9;">
        <li>Calculate your BMI as a baseline (<a href="/bmi-calculator" style="color:var(--accent);">use our free calculator</a>)</li>
        <li>Measure your waist circumference (aim: women &lt;80 cm, men &lt;94 cm)</li>
        <li>If BMI is borderline (23â€“27), consider measuring body fat percentage for more context</li>
        <li>For athletes or those who lift weights regularly, prioritise body fat % over BMI</li>
      </ol>
    </div>

    <h3>When BMI is enough</h3>
    <p>For most sedentary or lightly active adults, BMI is a reasonable and reliable screening tool. If your BMI is clearly in the healthy range and you don't have other risk factors, body fat measurement adds little additional information.</p>

    <h3>When you need body fat percentage</h3>
    <p>Consider measuring body fat if: you do regular resistance training; your BMI is borderline; you're over 50; or your doctor has flagged metabolic concerns despite a normal BMI.</p>

    <div class="warn-callout">
      <p>âš ï¸ Neither BMI nor body fat percentage is a clinical diagnosis. Both are screening tools. If you have concerns about your weight or metabolic health, consult a qualified healthcare provider who can assess your full picture â€” including blood pressure, blood glucose, lipid panel, and waist circumference.</p>
    </div>
  </div>

  <div class="article-section">
    <h2>The Bottom Line</h2>
    <p>BMI and body fat percentage each capture something real about health â€” but neither tells the whole story on its own. BMI is fast, free, and well-validated for population-level screening. Body fat percentage is more direct but harder to measure accurately.</p>
    <p>For most people, starting with BMI and adding waist circumference measurement gives you 80% of the health information you need. Body fat percentage is most valuable for athletes, older adults, and anyone whose BMI doesn't seem to match how they look or feel.</p>
  </div>

  <div class="faq-section">
  <h2 class="section-title">Frequently Asked Questions</h2>
  <div class="faq-item">
    <button class="faq-q" onclick="toggleFaq(this)">Which is more important: BMI or body fat percentage?</button>
    <div class="faq-a">Body fat percentage provides more clinically useful information because it directly measures what matters for health risk â€” the proportion of your body that is fat versus lean tissue.   </div>
  <div class="faq-item">
    <button class="faq-q" onclick="toggleFaq(this)">What body fat percentage is healthy?</button>
    <div class="faq-a">For men, 10â€“20% body fat is generally considered healthy (athletic 6â€“13%, fit 14â€“17%, acceptable 18â€“24%). For women, 18â€“28% is the healthy range (athletic 14â€“20%, fit 21â€“24%, acceptable 25â€“31%). These ranges vary slightly by age â€” body fat naturally increases with age.</div>
  </div>
  <div class="faq-item">
    <button class="faq-q" onclick="toggleFaq(this)">How do I measure body fat percentage at home?</button>
    <div class="faq-a">Navy method (waist, neck, and hip measurements) is accurate to within 3â€“4% and is free. Skinfold calipers used correctly are similar in accuracy.     <div class="faq-a">Yes â€” this is called 'normal weight obesity' or TOFI (Thin Outside, Fat Inside). A person with low muscle mass and moderate fat can have a normal BMI while having above-healthy body fat percentage. This pattern carries elevated metabolic risk despite normal BMI.</div>
  </div>
